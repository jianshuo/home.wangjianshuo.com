var ref = document.location.href;

if (ref.indexOf('?') != -1) {

qs = ref.substr(ref.indexOf('?')+1);

qsa = qs.split('&');

for (i=0;i<qsa.length;i++) {

qsip = qsa[i].split('=');

if (qsip.length == 1) continue;

if (qsip[0] == 'q' || qsip[0] == 'p') {

words = unescape(qsip[1].replace(/\+/g,' '));

try {

document.g.q.value = words;

}

catch(e) {}

}

}

}
