庆祝五一劳动节！劳动人民出身的我在家里闲着，就自己动手写了一个客齐集桌面包，只要把这个HTML文件存在自己的桌面上，就可以访问客齐集最新的上海的帖子了。

自己写代码总是可以丰衣足食。特别庆幸自己是技术出身，在互联网的世界里，或许商务很重要，或许资金很重要，但只有深切了解技术的人才能享受编点程序比吃个苹果还开心的那种心情

互联网归根到底，首先是技术的革命才带来了信息革命，才有了商业的革命。

如果谁想把这个东东放在自己的blog上面，只管看这个文件的源代码，拷贝粘贴过去就好。

google.load("feeds", "1");

function initialize() {
showfeed(0,20);
}

var g_feed = "http://shanghai.kijiji.cn/c2104.xml";
var g_current = 0;

function showfeed(start,end)
{
var feed = new google.feeds.Feed(g_feed);
feed.setNumEntries(end+20);
var loading = document.getElementById("loading");
loading.style.display = "block";
document.title += " > " + end ;
feed.load(function(result) {
loading.style.display = "none";
if (!result.error) {
var container = document.getElementById("feed");
while(container.firstChild) container.removeChild(container.firstChild);
for (var i = start; i < result.feed.entries.length && i <= end; i++) {
var div = document.createElement("div");
var div = document.createElement("div");
var entry = result.feed.entries[i];
var a = document.createElement("a");
a.setAttribute("href", entry.link);
a.setAttribute("title", entry.content);
a.setAttribute("style", "white-space: nowrap;");
a.onmouseover =  function (e) {
var a;
if(!e) a = window.event.srcElement;
else a = e.target;
var container = document.getElementById("body");
container.innerHTML = "<h3>" + a.innerHTML + "</h3>" + a.title.replace(/&lt;br \/&gt;/g, "<br>");
}
a.setAttribute("target", "_blank");
a.innerHTML = entry.title;
div.appendChild(a);
container.appendChild(div);
}
if(result.feed.entries.length < end)
container.innerHTML = "更多信息，请登录<a target=_blank href=" + g_feed.replace(".xml", "") + ">客齐集</a>";
} else document.title = "错误：" + result.error.code + ": " + result.error.message;

});
}
google.setOnLoadCallback(initialize);

function showcontent(a) {
var container = document.getElementById("body");
container.innerHTML = "<h3>" + a.innerHTML + "</h3>" + a.title.replace(/&lt;br \/&gt;/g, "<br>");
}

function gonext(change) {
if(g_current+change >= 0)
g_current += change;

showfeed(g_current, g_current + 20);
}

function changefeed(e) {
g_feed = e.title;
g_current = 0;
showfeed(0,20);
var children = e.parentNode.childNodes;
for (var i = 0; i < children.length; i++)
if(children[i].tagName=="A")
children[i].className = "normal";
e.className = "highlight";
document.title = "客齐集桌面包 - " + e.innerHTML;
}

.highlight {background-color:#ddddff;padding:1px;}
.normal {background-color:white;padding:1px;}
