function applyTag(){

var s = document.getElementById('tags').value;

$.each(s.trim().split(','), function(key, value){

if(value == '') return;

$(".datagrid li:contains('" + value + "')").appendTo($('#completed')).css('color', 'red').append(" <span>" + value + "</span>");

});

}
